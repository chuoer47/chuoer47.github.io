def judge(case):
    """
    判断前case个是否满足借教室的条件，满足返回False，不满足Ture
    :param case:
    :return:
    """
    # 初始化差分数组
    for i in range(len(darr)):
        darr[i] = 0
    # 利用差分数组性质
    for i in range(0, case):
        d, s, j = arr[i]
        darr[s] -= d
        darr[j + 1] += d
    # 还原原始数组
    tem = 0
    for i in range(1, n + 1):
        tem += darr[i]
        if classroom[i] + tem < 0:
            return True
    return False

if not judge(m):
    print(0)
    exit(0)
# 发现满足不了，二分法找到第一个出现问题的条件

r = m

    if judge(mid):
        r = mid

def cal(lst: list, case: int):
    a, b = min(lst), max(lst),
    if case > a:
        return 0
    return (a // case) * (b // case)

def judge(case):
    """
    判断case时刻是否都有水流，有返回TRUE,无返回FALSE
    :param case:
    :return:
    """
    # 初始化状态数组
    state = []
    # 确定水流
    for i in range(0, n):
        if case < S[i]:  # 还没有开闸
            continue
        l, r = sgn(L[i] - (case - S[i])), sgn(L[i] + (case - S[i]))
        state.append((l, r))
    state.sort()
    # print(case,state)

    if l != 1:
        return False
    for i in range(1, len(state)):
        ll, rr = state[i]
        if ll > r+1:
            return False
        else:
            r = max(rr, r)
    return (l == 1) and (r == length)

def judge(x):
    """
    该函数返回大于等于x的数量s是否大于k
    当s>m,返回False，反之，返回Ture
    :param x:
    :return:
    """
    s = 0

        a, b = arr[i]

            s = s + ((a - x) // b) + 1

https://www.acwing.com/problem/content/104/

def check(case):
    """
    用来判断在arr中平均值最大能否达到case
    能达到返回TRUE,不嫩返回FALSE
    :param case:
    :return:
    """
    # 构造前缀和数组
    for i in range(1, n + 1):
        d_arr[i] = d_arr[i - 1] + arr[i] - case
    # 双指针检验
    left, right = 0, f
    min_left = d_arr[left]
    while right <= n:
        min_left = min(min_left, d_arr[left])
        if d_arr[right] - min_left >= 0:
            return True
        left += 1
        right += 1
    return False

n, f = map(int, input().split())
arr = [0 for _ in range(0, n + 10)]
for i in range(1, n + 1):
    arr[i] = int(input())
d_arr = [0 for _ in range(0, n + 10)]  # 前缀和数组
#   使用二分法寻找最大值，这个设计很巧妙
l, r = 0, max(arr)
while r - l > 1e-6:
    mid = (r + l) / 2
    if check(mid):
        l = mid
    else:
        r = mid
